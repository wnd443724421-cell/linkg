  840 /**
  841  * @brief 判断Wire窗口是否达到BAD丢包阈值。
  842  *
  843  * BAD要求loss >= max(0.5%, 50% / sqrt(N))。
  844  * 等价整数条件：200 * L >= N && 4 * L * L >= N。
  845  * 调用方保证N >= 20，且L <= N。
  846  */
  847 static bool _linkg_switch_quality_wire_bad(uint64_t samples, uint64_t lost_packets)
  848 {
  849     // 使用除法计算ceil(N / 200)，避免乘法溢出。
  850     if (lost_packets < samples / 200ULL + (samples % 200ULL != 0U ? 1ULL : 0ULL))
  851     {
  852         return false;
  853     }
  854 
  855     // 只有L >= 2^31时4*L*L才可能超过uint64_t范围，此时必定满足第二条件。
  856     return lost_packets >= 2147483648ULL || 4ULL * lost_packets * lost_packets >= samples;
  857 }
  858 
  859 /**
  860  * @brief 判断Wire窗口是否达到GOOD丢包阈值。
  861  *
  862  * GOOD要求loss <= min(1%, BAD阈值的一半)。
  863  * N < 10000：100 * L <= N && 16 * L * L <= N。
  864  * N >= 10000：400 * L <= N，即丢包率不超过0.25%。
  865  */
  866 static bool _linkg_switch_quality_wire_good(uint64_t samples, uint64_t lost_packets)
  867 {
  868     if (samples >= 10000ULL)
  869     {
  870         return lost_packets <= UINT64_MAX / 400ULL && 400ULL * lost_packets <= samples;
  871     }
  872 
  873     return 100ULL * lost_packets <= samples && 16ULL * lost_packets * lost_packets <= samples;
  874 }
